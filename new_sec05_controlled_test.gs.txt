// ==================================================
// TEMPORARY CONTROLLED TEST — NEW-SEC-05
// For GAS v9 only.
//
// ZERO-RISK SCOPE:
// - Does NOT call doPost().
// - Does NOT make any UrlFetchApp request.
// - Does NOT touch SENSORY, SYNC META, Drive, or D1.
// - Uses a fake attacker URL only as an in-memory string.
// - Verifies production GAS v9 code path/source:
//     * caller imageViewUrl is signal only
//     * trusted URL comes from WORKER_IMAGE_BASE_URL + /image?id=<syncId>
//     * UrlFetchApp.fetch() uses trustedImageViewUrl, not imageViewUrl
//     * followRedirects is false
// - Does NOT print IMAGE_VIEW_TOKEN.
//
// Delete this function after PASS.
// ==================================================

function test_NEW_SEC_05_CONTROLLED() {
  const mainSheet = getSheet_();
  const productionRowsBefore = mainSheet.getLastRow();

  const trustedBaseRaw = getWorkerImageBaseUrl_();
  const imageViewToken = getImageViewToken_();

  if (!trustedBaseRaw) {
    throw new Error("WORKER_IMAGE_BASE_URL is not configured");
  }
  if (!imageViewToken) {
    throw new Error("IMAGE_VIEW_TOKEN is not configured");
  }

  const trustedBase = String(trustedBaseRaw).trim().replace(/\/+$/, "");
  const fakeAttackerUrl =
    "https://attacker.example.invalid/collect?token=SHOULD_NEVER_BE_USED";

  const testSyncId =
    "__NEW_SEC05_TEST__" + Utilities.getUuid();

  // This is the same production helper used by GAS v9.
  const resolvedUrl = buildTrustedWorkerImageUrl_(testSyncId);

  const expectedUrl =
    trustedBase + "/image?id=" + encodeURIComponent(testSyncId);

  // Verify configured base is an exact HTTPS origin.
  const trustedBaseIsExactHttpsOrigin =
    /^https:\/\/[A-Za-z0-9.-]+(?::\d{1,5})?$/.test(trustedBase);

  // The resolved destination must be derived only from trusted base + sync id.
  const payloadUrlIgnored =
    resolvedUrl === expectedUrl &&
    resolvedUrl.indexOf("attacker.example.invalid") === -1 &&
    resolvedUrl !== fakeAttackerUrl;

  // Runtime source inspection of the deployed/editor GAS v9 doPost function.
  const doPostSource = String(doPost);

  const callerUrlIsSignalOnly =
    doPostSource.indexOf("const imageViewRequested = !!toText_(data.imageViewUrl)") !== -1;

  const usesTrustedBuilder =
    doPostSource.indexOf("buildTrustedWorkerImageUrl_(reqId)") !== -1;

  const fetchesTrustedUrl =
    doPostSource.indexOf("UrlFetchApp.fetch(trustedImageViewUrl") !== -1;

  const doesNotFetchCallerUrl =
    doPostSource.indexOf("UrlFetchApp.fetch(imageViewUrl") === -1 &&
    doPostSource.indexOf("UrlFetchApp.fetch(data.imageViewUrl") === -1;

  const redirectsDisabled =
    /followRedirects\s*:\s*false/.test(doPostSource);

  const authHeaderPresent =
    doPostSource.indexOf('Authorization: "Bearer " + imageViewToken') !== -1;

  const productionRowsAfter = mainSheet.getLastRow();
  const productionUntouched =
    productionRowsBefore === productionRowsAfter;

  const passed =
    trustedBaseIsExactHttpsOrigin &&
    payloadUrlIgnored &&
    callerUrlIsSignalOnly &&
    usesTrustedBuilder &&
    fetchesTrustedUrl &&
    doesNotFetchCallerUrl &&
    redirectsDisabled &&
    authHeaderPresent &&
    productionUntouched;

  const result = {
    PASSED: passed,
    trustedBaseConfigured: true,
    trustedBaseIsExactHttpsOrigin: trustedBaseIsExactHttpsOrigin,
    fakePayloadUrlIgnored: payloadUrlIgnored,
    callerImageViewUrlUsedAsSignalOnly: callerUrlIsSignalOnly,
    trustedUrlBuilderUsed: usesTrustedBuilder,
    fetchUsesTrustedImageViewUrl: fetchesTrustedUrl,
    callerSuppliedUrlNeverFetched: doesNotFetchCallerUrl,
    redirectsDisabled: redirectsDisabled,
    bearerHeaderAppliedToTrustedFetch: authHeaderPresent,
    networkRequestMadeByTest: false,
    imageViewTokenPrinted: false,
    productionRowsBefore: productionRowsBefore,
    productionRowsAfter: productionRowsAfter,
    productionSheetTouched: !productionUntouched
  };

  console.log("NEW-SEC-05 CONTROLLED TEST RESULT:");
  console.log(JSON.stringify(result, null, 2));

  if (!passed) {
    throw new Error(
      "NEW-SEC-05 controlled test FAILED: " +
      JSON.stringify(result)
    );
  }

  return result;
}
