// ==================================================
// TEMPORARY CONTROLLED TEST v2 — SEC-02 / DELETE_DRIVE
// GAS v7 production code is NOT changed.
//
// Difference from v1:
// Re-fetches the Drive file after DELETE_DRIVE before checking isTrashed(),
// avoiding a stale in-memory Drive File object false-negative.
// ==================================================

function test_SEC02_DELETE_DRIVE_CONTROLLED_V2() {
  const mainSheet = getSheet_();
  const ss = mainSheet.getParent();
  const metaSheet = ensureSyncMetaSheet_(mainSheet);

  const productionRowsBefore = mainSheet.getLastRow();

  const deleteToken = getDeleteDriveToken_();
  const syncToken = getSyncToken_();

  if (!deleteToken) throw new Error("DELETE_DRIVE_TOKEN is not configured");
  if (!syncToken) throw new Error("SYNC_TOKEN is not configured");

  const testId = "__SEC02_DELETE_TEST_V2__" + Utilities.getUuid();
  const secureFolder = DriveApp.getFolderById(DRIVE_FOLDER_ID);

  let secureFileA = null;
  let secureFileB = null;
  let outsideFile = null;

  function callDoPost_(payload) {
    const res = doPost({
      postData: {
        contents: JSON.stringify(payload)
      }
    });
    return JSON.parse(res.getContent());
  }

  function safeTrashById_(fileId) {
    if (!fileId) return;
    try {
      const f = DriveApp.getFileById(fileId);
      if (!f.isTrashed()) f.setTrashed(true);
    } catch (e) {}
  }

  try {
    secureFileA = secureFolder.createFile(
      "SEC02_TEST_V2_A_" + Utilities.getUuid() + ".txt",
      "TEMP SEC-02 TEST A"
    );

    secureFileB = secureFolder.createFile(
      "SEC02_TEST_V2_B_" + Utilities.getUuid() + ".txt",
      "TEMP SEC-02 TEST B"
    );

    outsideFile = DriveApp.createFile(
      "SEC02_TEST_V2_OUTSIDE_" + Utilities.getUuid() + ".txt",
      "TEMP SEC-02 TEST OUTSIDE"
    );

    const fileAId = secureFileA.getId();
    const fileBId = secureFileB.getId();
    const outsideId = outsideFile.getId();

    appendSyncMeta_(
      metaSheet,
      testId,
      fileAId,
      "",
      formatDateTimeDMY(new Date())
    );

    // 1) SYNC_TOKEN cannot authorize DELETE_DRIVE
    const t1 = callDoPost_({
      token: syncToken,
      action: "DELETE_DRIVE",
      fileIds: [fileBId]
    });

    const test1Pass =
      t1 &&
      t1.success === false &&
      String(t1.message || "") === "Unauthorized" &&
      DriveApp.getFileById(fileBId).isTrashed() === false;

    // 2) Wrong DELETE_DRIVE_TOKEN rejected
    const t2 = callDoPost_({
      token: "__WRONG_DELETE_DRIVE_TOKEN__",
      action: "DELETE_DRIVE",
      fileIds: [fileBId]
    });

    const test2Pass =
      t2 &&
      t2.success === false &&
      String(t2.message || "") === "Unauthorized" &&
      DriveApp.getFileById(fileBId).isTrashed() === false;

    // 3) Outside-folder file rejected
    const t3 = callDoPost_({
      token: deleteToken,
      action: "DELETE_DRIVE",
      fileIds: [outsideId]
    });

    const t3Reason =
      t3 &&
      Array.isArray(t3.failedDetails) &&
      t3.failedDetails.length
        ? String(t3.failedDetails[0].reason || "")
        : "";

    const test3Pass =
      t3 &&
      t3.success === true &&
      Number(t3.deletedCount || 0) === 0 &&
      t3Reason === "DRIVE_FILE_OUTSIDE_SECURE_ENTRY_FOLDER" &&
      DriveApp.getFileById(outsideId).isTrashed() === false;

    // 4) syncId -> Drive ID mismatch rejected
    const t4 = callDoPost_({
      token: deleteToken,
      action: "DELETE_DRIVE",
      items: [{
        syncId: testId,
        driveFileId: fileBId,
        requireDrive: true
      }]
    });

    const t4Reason =
      t4 &&
      Array.isArray(t4.failedDetails) &&
      t4.failedDetails.length
        ? String(t4.failedDetails[0].reason || "")
        : "";

    const metaAfterMismatch = findSyncRecordById_(metaSheet, testId);

    const test4Pass =
      t4 &&
      t4.success === true &&
      Number(t4.deletedCount || 0) === 0 &&
      t4Reason === "DRIVE_FILE_ID_MISMATCH" &&
      DriveApp.getFileById(fileAId).isTrashed() === false &&
      DriveApp.getFileById(fileBId).isTrashed() === false &&
      metaAfterMismatch &&
      String(metaAfterMismatch.driveFileId || "") === fileAId;

    // 5) Valid deletion
    const t5 = callDoPost_({
      token: deleteToken,
      action: "DELETE_DRIVE",
      items: [{
        syncId: testId,
        driveFileId: fileAId,
        requireDrive: true
      }]
    });

    // Important v2 fix: fetch a NEW Drive object after deletion.
    Utilities.sleep(300);
    let fileATrashedAfter = false;
    try {
      fileATrashedAfter = DriveApp.getFileById(fileAId).isTrashed() === true;
    } catch (e) {
      // If Google no longer resolves a trashed temporary file, that also
      // confirms it is no longer active/accessible.
      fileATrashedAfter = true;
    }

    const metaAfterValidDelete = findSyncRecordById_(metaSheet, testId);

    const confirmedContainsId =
      t5 &&
      Array.isArray(t5.confirmedIds) &&
      t5.confirmedIds.indexOf(testId) !== -1;

    const test5Pass =
      t5 &&
      t5.success === true &&
      Number(t5.deletedCount || 0) === 1 &&
      confirmedContainsId &&
      fileATrashedAfter &&
      metaAfterValidDelete === null;

    const productionRowsAfter = mainSheet.getLastRow();
    const productionUntouched = productionRowsBefore === productionRowsAfter;

    const passed =
      test1Pass &&
      test2Pass &&
      test3Pass &&
      test4Pass &&
      test5Pass &&
      productionUntouched;

    const result = {
      PASSED: passed,
      syncTokenCannotDeleteDrive: test1Pass,
      wrongDeleteTokenRejected: test2Pass,
      outsideFolderRejected: test3Pass,
      syncIdDriveIdMismatchRejected: test4Pass,
      validSecureEntryDeleteConfirmed: test5Pass,
      validDeleteResponseSuccess: !!(t5 && t5.success === true),
      validDeleteDeletedCount: t5 ? Number(t5.deletedCount || 0) : null,
      validDeleteConfirmedId: confirmedContainsId,
      validDeleteFileTrashedAfterRefetch: fileATrashedAfter,
      validDeleteMetaRemoved: metaAfterValidDelete === null,
      productionRowsBefore: productionRowsBefore,
      productionRowsAfter: productionRowsAfter,
      productionSheetTouched: !productionUntouched,
      testTokensPrinted: false
    };

    console.log("SEC-02 DELETE_DRIVE CONTROLLED TEST V2 RESULT:");
    console.log(JSON.stringify(result, null, 2));

    if (!passed) {
      throw new Error(
        "SEC-02 DELETE_DRIVE controlled test V2 FAILED: " +
        JSON.stringify(result)
      );
    }

    return result;

  } finally {
    try {
      deleteSyncMetaById_(metaSheet, testId);
    } catch (e) {}

    try { safeTrashById_(secureFileA ? secureFileA.getId() : ""); } catch (e) {}
    try { safeTrashById_(secureFileB ? secureFileB.getId() : ""); } catch (e) {}
    try { safeTrashById_(outsideFile ? outsideFile.getId() : ""); } catch (e) {}
  }
}
